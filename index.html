function switchTab(evt, tabName) { var i, tabcontent, tablinks; tabcontent = document.getElementsByClassName("tab-content"); for (i = 0; i < tabcontent.length; i++) { tabcontent[i].classList.remove("active"); } tablinks = document.getElementsByClassName("tab-button"); for (i = 0; i < tablinks.length; i++) { tablinks[i].classList.remove("active"); } document.getElementById(tabName).classList.add("active"); evt.currentTarget.classList.add("active"); } function resetFinalPrice(type) { if (type === 'percentage') { document.getElementById("finalPriceInput_pct").value = ""; calculatePercentage(); } else { document.getElementById("finalPriceInput_fr").value = ""; calculateFullReduction(); } } function saveConfig(type) { if (type === 'percentage') { localStorage.setItem("officialDiscount_pct", document.getElementById("officialDiscount_pct").value); localStorage.setItem("storeStep_pct", document.getElementById("storeStep_pct").value); localStorage.setItem("storeDiscount_pct", document.getElementById("storeDiscount_pct").value); } else { localStorage.setItem("officialStep_fr", document.getElementById("officialStep_fr").value); localStorage.setItem("officialDiscount_fr", document.getElementById("officialDiscount_fr").value); localStorage.setItem("storeStep_fr", document.getElementById("storeStep_fr").value); localStorage.setItem("storeDiscount_fr", document.getElementById("storeDiscount_fr").value); } } function loadConfig(type) { if (type === 'percentage') { document.getElementById("officialDiscount_pct").value = localStorage.getItem("officialDiscount_pct") || "15"; document.getElementById("storeStep_pct").value = localStorage.getItem("storeStep_pct") || "1000"; document.getElementById("storeDiscount_pct").value = localStorage.getItem("storeDiscount_pct") || "50"; calculatePercentage(); } else { document.getElementById("officialStep_fr").value = localStorage.getItem("officialStep_fr") || "300"; document.getElementById("officialDiscount_fr").value = localStorage.getItem("officialDiscount_fr") || "30"; document.getElementById("storeStep_fr").value = localStorage.getItem("storeStep_fr") || "1000"; document.getElementById("storeDiscount_fr").value = localStorage.getItem("storeDiscount_fr") || "50"; calculateFullReduction(); } } window.onload = function() { loadConfig('percentage'); loadConfig('fullReduction'); };
